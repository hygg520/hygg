local function setCharacterTransparency(character:Model,transparency:number):()
	for _,descendant in character:GetDescendants() do
		if descendant:IsA("BasePart") or descendant:IsA("Decal") then
			descendant.Transparency=transparency
		end
	end
end
local function getHumanoid():Humanoid?
	local character=player.Character
	if not character then return nil end
	return character:FindFirstChild("Humanoid")::Humanoid?
end
local function getHumanoidRootPart():BasePart?
	local character=player.Character
	if not character then return nil end
	return character:FindFirstChild("HumanoidRootPart")::BasePart?
end
local function startAntiPullback():()
	if antiPullbackConn then antiPullbackConn:Disconnect() antiPullbackConn=nil end
	antiPullbackConn=RunService.Heartbeat:Connect(function()
		local character=player.Character
		if not character then return end
		local humanoid=character:FindFirstChildOfClass("Humanoid")
		local root=character:FindFirstChild("HumanoidRootPart")
		if not humanoid or not root then return end
		if humanoid.MoveDirection.Magnitude>0 then
			character:TranslateBy(humanoid.MoveDirection*CONFIG.ANTI_PULLBACK_SPEED/20)
		end
	end)
end
local function stopAntiPullback():()
	if antiPullbackConn then antiPullbackConn:Disconnect() antiPullbackConn=nil end
end
local function toggleInvisibility():()
	if not player.Character then return end
	playerState.isInvisible=not playerState.isInvisible
	if playerState.isInvisible then
		local humanoidRootPart=getHumanoidRootPart()
		if not humanoidRootPart then return end
		local savedPosition=humanoidRootPart.CFrame
		player.Character:MoveTo(CONFIG.INVISIBILITY_POSITION)
		task.wait(0.15)
		local seat=Instance.new("Seat")
		seat.Name="invischair"
		seat.Anchored=false
		seat.CanCollide=false
		seat.Transparency=1
		seat.Position=CONFIG.INVISIBILITY_POSITION
		seat.Parent=workspace
		local weld=Instance.new("Weld")
		weld.Part0=seat
		weld.Part1=player.Character:FindFirstChild("Torso") or player.Character:FindFirstChild("UpperTorso")
		weld.Parent=seat
		task.wait()
		seat.CFrame=savedPosition
		setCharacterTransparency(player.Character,0.5)
		InvisButton.Text="显示"
	else
		local invisChair=workspace:FindFirstChild("invischair")
		if invisChair then invisChair:Destroy() end
		if player.Character then setCharacterTransparency(player.Character,0) end
		InvisButton.Text="隐身"
	end
end
local function toggleSpeedBoost():()
	playerState.isSpeedBoosted=not playerState.isSpeedBoosted
	if playerState.isSpeedBoosted then
		startAntiPullback()
		SpeedButton.Text="关闭加速"
	else
		stopAntiPullback()
		SpeedButton.Text="加速"
	end
end
local function resetPlayerState():()
	playerState.isInvisible=false
	playerState.isSpeedBoosted=false
	stopAntiPullback()
	if InvisButton then InvisButton.Text="隐身" end
	if SpeedButton then SpeedButton.Text="加速" end
end
InvisButton.MouseButton1Click:Connect(toggleInvisibility)
SpeedButton.MouseButton1Click:Connect(toggleSpeedBoost)
CloseButton.MouseButton1Click:Connect(function()
	ScreenGui.Enabled=false
end)
local minimized=false
MinimizeButton.MouseButton1Click:Connect(function()
	minimized=not minimized
	if minimized then
		Frame.Size=UDim2.new(0,158,0,25)
		InvisButton.Visible=false
		SpeedButton.Visible=false
		MinimizeButton.Text="+"
	else
		Frame.Size=UDim2.new(0,158,0,110)
		InvisButton.Visible=true
		SpeedButton.Visible=true
		MinimizeButton.Text="-"
	end
end)
UserInputService.InputBegan:Connect(function(input:InputObject,gameProcessed:boolean)
	if gameProcessed then return end
	if input.KeyCode==CONFIG.TOGGLE_KEY then toggleInvisibility() end
end)
player.CharacterAdded:Connect(function(character:Model)
	resetPlayerState()
	local humanoid=character:WaitForChild("Humanoid")::Humanoid
	playerState.originalSpeed=humanoid.WalkSpeed
	local invisChair=workspace:FindFirstChild("invischair")
	if invisChair then invisChair:Destroy() end
end)
Players.PlayerRemoving:Connect(function(leavingPlayer:Player)
	if leavingPlayer==player then
		local invisChair=workspace:FindFirstChild("invischair")
		if invisChair then invisChair:Destroy() end
		stopAntiPullback()
	end
end)
