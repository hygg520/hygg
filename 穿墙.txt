local Players = cloneref and cloneref(game:GetService("Players")) or game:GetService("Players") local RunService = cloneref and cloneref(game:GetService("RunService")) or game:GetService("RunService") local LocalPlayer = Players.LocalPlayer local function enableNoclip() local char = LocalPlayer.Character if not char then return end for _, part in ipairs(char:GetDescendants()) do if part:IsA("BasePart") then pcall(function() part.CanCollide = false end) end end end local connection connection = RunService.Stepped:Connect(function() enableNoclip() end) if not connection then connection = RunService.Heartbeat:Connect(function() enableNoclip() end) end
